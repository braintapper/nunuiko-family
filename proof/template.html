Texture healing

Next to a narrow letter (i j l r t f I), a wide letter (m w M W A O Q V …) borrows up to 60 units of its space, and the narrow letter steps aside. Cells stay 600 wide, so columns still line up. It is the calt feature; top row on, bottom row off.

import { readFile } from "node:fs/promises";

// Parse a config file; fall back to defaults on any error.
export async function loadConfig(path: string): Promise<Config> {
  try {
    const raw = await readFile(path, "utf8");
    const cfg = JSON.parse(raw) as Partial<Config>;
    return { ...defaults, ...cfg, retries: cfg.retries ?? 3 };
  } catch (err) {
    console.warn(`config: ${path} unreadable (${err})`);
    return defaults;
  }
}

Python

from dataclasses import dataclass


@dataclass(frozen=True)
class Reading:
    sensor: str
    value: float
    ok: bool = True


def average(readings: list[Reading], floor: float = 0.0) -> float | None:
    """Mean of the good readings at or above floor, or None."""
    good = [r.value for r in readings if r.ok and r.value >= floor]
    return sum(good) / len(good) if good else None  # 0 readings -> None
